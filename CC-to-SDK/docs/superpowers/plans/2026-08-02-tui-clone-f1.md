- Produces:

  ```ts
  export type TranscriptSource = "host" | "disk" | "local";
  export type SidecarAssociation =
    | { scope: "call"; value: unknown }
    | { scope: "message"; value: unknown; reason: "no-tool-result" | "missing-tool-use-id" | "unmatched-tool-use" | "ambiguous-tool-results" | "duplicate-tool-use" | "duplicate-tool-result" };
  import type { RenderLine } from "./render.js";
  export type LocalTranscriptEvent = {
    kind: "notice" | "user-echo" | "command-echo" | "resume-divider" | "rewind-divider" | "source-gap" | "decision-settled" | "visual";
    lines: readonly RenderLine[];
    data?: Record<string, unknown>;
  };
  export type TranscriptBootstrapEntry =
    | { kind: "sdk"; source: "disk"; message: Record<string, unknown> }
    | { kind: "local"; identity: string; event: LocalTranscriptEvent };
  export type TranscriptEntry =
    | { sequence: number; source: "host" | "disk"; kind: "sdk-message"; identity?: string; message: Record<string, unknown>; sidecar?: SidecarAssociation }
    | { sequence: number; source: "local"; kind: "local-event"; identity: string; event: LocalTranscriptEvent };
  export interface ToolEvent {
    id: string; name: string; input: unknown; callSequence: number;
    route: "top-level" | "nested"; parent_tool_use_id?: string;
    result?: { content: unknown; isError: boolean; resultSequence: number; sidecar?: SidecarAssociation };
  }
  export class TranscriptDocument {
    appendSdk(source: "host" | "disk", message: Record<string, unknown>): boolean;
    appendLocal(event: LocalTranscriptEvent, identity: string): boolean;
    appendFollowGap(identity: string): boolean;
    entries(): readonly TranscriptEntry[];
    toolEvents(): readonly ToolEvent[];
  }
  export type ToolStatus = "running" | "success" | "error" | "interrupted" | "rejected" | "suppressed";
  export interface NormalizedToolResult {
    tool: string; status: ToolStatus; source: "pending" | "structured" | "fallback";
    rawContent: unknown; flatText: string; summary: string; output: string; outputLines: readonly string[]; structured?: Record<string, unknown>;
  }
  export function normalizeToolResult(event: ToolEvent, options?: { verbose?: boolean }): NormalizedToolResult;
  ```

  Task 1 is renderer-independent: local records are `LocalTranscriptEvent`, never a future `RenderItem`, so its typecheck does not depend on Task 3. `RenderLine[]` is permitted only as the faithful payload of an inherently local visual event: it retains existing `/help`, `/usage`, `/status`, local Bash, welcome, notice, and error formatter output without a second ordered transcript or formatter rewrite. It is never the canonical reduction of an SDK message or tool result. `appendSdk` uses only a source-stable identity to deduplicate disk/follow repetition: user `uuid`, assistant `message.id`, or a tool-use ID tuple. If no such identity exists, append rather than structurally hashing payload text; two equal-looking calls can be distinct turns. This preserves truth rather than hiding a genuine repeated call. `appendFollowGap()` itself stores `{ kind: "source-gap", lines: [{ text: "Earlier live output unavailable while attaching", dim: true }] }`; callers supply only its stable identity, and Task 4 calls it only on a truncated follow-start event. `TranscriptBootstrapEntry[]` is the sole startup handoff: array order is the total SDK/local order, and every local envelope already carries the stable identity reused by projection. `appendLocal` deduplicates only by that identity, never by line payload; equal-looking events with different identities are distinct history.
